## Implementação Completa

### JavaScript SDK

```javascript
class SocialMediaVideoCreator {
  constructor(apiUrl = 'http://localhost:3000') {
    this.apiUrl = apiUrl;
  }
  
  async createInstagramStory(config) {
    const timeline = {
      tracks: [
        {
          clips: [
            // Background
            {
              asset: {
                type: "image",
                source: "url",
                src: config.backgroundImage
              },
              start: 0,
              length: config.duration || 15
            },
            // Main content
            {
              asset: {
                type: "video",
                source: "url",
                src: config.mainVideo
              },
              start: 2,
              length: (config.duration || 15) - 4
            },
            // Title
            {
              asset: {
                type: "text",
                text: config.title,
                style: {
                  fontSize: 48,
                  fontColor: "white",
                  bold: true,
                  alignment: "center",
                  position: "top",
                  outlineColor: "black",
                  marginV: 100
                }
              },
              start: 0,
              length: 3
            },
            // CTA
            {
              asset: {
                type: "text",
                text: config.cta || "Swipe up para saber mais",
                style: {
                  fontSize: 32,
                  fontColor: "white",
                  alignment: "center",
                  position: "bottom",
                  outlineColor: "black",
                  marginV: 150
                }
              },
              start: (config.duration || 15) - 3,
              length: 3
            }
          ]
        }
      ]
    };
    
    if (config.backgroundMusic) {
      timeline.tracks.push({
        clips: [{
          asset: {
            type: "audio",
            source: "url",
            src: config.backgroundMusic,
            isBackground: true,
            volume: 0.2
          },
          start: 0,
          length: config.duration || 15
        }]
      });
    }
    
    return await this.render(timeline, {
      format: "mp4",
      resolution: "1080x1920",
      quality: "high",
      fps: 30
    });
  }
  
  async createTikTokVideo(config) {
    const timeline = {
      tracks: [{
        clips: [
          // Main video
          {
            asset: {
              type: "video",
              source: "url",
              src: config.mainVideo
            },
            start: 0,
            length: config.duration || 30
          },
          // Title
          {
            asset: {
              type: "text",
              text: config.title,
              style: {
                fontSize: 54,
                fontColor: "yellow",
                bold: true,
                alignment: "center",
                position: "top",
                outlineColor: "black",
                marginV: 80
              }
            },
            start: 0,
            length: 5
          }
        ]
      }]
    };
    
    // Add step-by-step texts
    if (config.steps) {
      config.steps.forEach((step, index) => {
        const stepDuration = (config.duration || 30) / config.steps.length;
        timeline.tracks[0].clips.push({
          asset: {
            type: "text",
            text: step,
            style: {
              fontSize: 36,
              fontColor: "white",
              alignment: "left",
              position: "bottom",
              outlineColor: "black",
              marginV: 200
            }
          },
          start: 5 + (index * stepDuration),
          length: stepDuration
        });
      });
    }
    
    return await this.render(timeline, {
      format: "mp4",
      resolution: "1080x1920",
      quality: "high",
      fps: 30
    });
  }
  
  async createSquarePost(config) {
    const timeline = {
      tracks: [{
        clips: [
          // Background
          {
            asset: {
              type: "image",
              source: "url",
              src: config.backgroundImage
            },
            start: 0,
            length: config.duration || 10
          },
          // Title
          {
            asset: {
              type: "text",
              text: config.title,
              style: {
                fontSize: 48,
                fontColor: "white",
                bold: true,
                alignment: "center",
                position: "top",
                outlineColor: "black",
                marginV: 100
              }
            },
            start: 0,
            length: config.duration || 10
          }
        ]
      }]
    };
    
    // Add offer text if provided
    if (config.offer) {
      timeline.tracks[0].clips.push({
        asset: {
          type: "text",
          text: config.offer,
          style: {
            fontSize: 72,
            fontColor: "red",
            bold: true,
            alignment: "center",
            position: "center"
          }
        },
        start: 3,
        length: 4
      });
    }
    
    return await this.render(timeline, {
      format: "mp4",
      resolution: "1080x1080",
      quality: "high",
      fps: 30
    });
  }
  
  async render(timeline, output) {
    const response = await fetch(`${this.apiUrl}/api/render`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        timeline,
        output
      })
    });
    
    const result = await response.json();
    
    if (!response.ok) {
      throw new Error(result.error?.message || 'Erro ao renderizar vídeo');
    }
    
    return result;
  }
  
  async waitForCompletion(jobId) {
    while (true) {
      const response = await fetch(`${this.apiUrl}/api/media/job/${jobId}/status`);
      const status = await response.json();
      
      if (status.status === 'completed') {
        return status;
      } else if (status.status === 'failed') {
        throw new Error(status.error?.message || 'Job failed');
      }
      
      // Wait 2 seconds before checking again
      await new Promise(resolve => setTimeout(resolve, 2000));
    }
  }
}
```

```javascript
const creator = new SocialMediaVideoCreator();

// Instagram Story
```

## Dicas de Otimização

### Performance

1. **Use resoluções otimizadas** para cada plataforma
2. **Mantenha durações apropriadas** para engagement
3. **Comprima áudios** para reduzir tamanho final
4. **Use qualidade 'medium'** para redes sociais (suficiente)

### Engagement

1. **Textos grandes e legíveis** em dispositivos móveis
2. **Cores contrastantes** para destacar elementos
3. **Call-to-actions claros** e posicionados estrategicamente
4. **Música de fundo** adequada ao conteúdo

```javascript
  ];
  
  const jobs = await Promise.all(
```
